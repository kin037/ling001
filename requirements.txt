    report = f"""

- **现货价：** {data['prices']['smm']} 万元/吨
- **期货价：** {data['prices']['futures']} 万元/吨

    return report

def push_to_wechat(content):
    """推送到微信"""
    if not PUSHPLUS_TOKEN:
        print("⚠️ 未检测到 Token，跳过推送，直接在下方显示报告：")
        print("-" * 30)
        print(content) # 直接在手机屏幕上打印出来
        return

    url = "https://www.pushplus.plus/send"
    payload = {
        "token": PUSHPLUS_TOKEN,
        "title": "锂矿情报日报",
        "content": content,
        "template": "markdown"
    }
    
    print("🚀 正在发送微信推送...")
    try:
        # 设置 timeout=10 防止手机网络卡死程序
        resp = requests.post(url, json=payload, timeout=10)
        
        if resp.status_code == 200:
            result = resp.json()
            if result.get("code") == 200:
                print("✅ 推送成功！请查收微信。")
            else:
                print(f"❌ 推送接口报错: {result.get('msg')}")
        else:
            print(f"❌ 网络请求失败，状态码: {resp.status_code}")
            
    except requests.exceptions.RequestException as e:
        print(f"❌ 网络连接超时或错误: {e}")
        print("💡 提示：请检查手机网络或 VPN 设置")

# ==========================================
# ▶️ 启动入口
# ==========================================
if __name__ == "__main__":
    data = fetch_data()
    if data:
        report = generate_report(data)
        push_to_wechat(report)
    else:
        print("⛔ 因数据获取失败，程序终止。")
